cmake_minimum_required(VERSION 3.15)
project(SITRAC LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MODELO src/interfaz.cpp src/util.cpp src/ciudades.cpp src/rutas.cpp src/pasajeros.cpp src/cola.cpp src/historial.cpp src/busquedas.cpp src/reportes.cpp src/archivos.cpp)
if(WIN32)
  enable_language(RC)
  add_executable(SITRAC WIN32 src/windows.cpp windows.rc ${MODELO})
  target_link_libraries(SITRAC PRIVATE comctl32 comdlg32 gdi32)
  if(MINGW)
    target_link_options(SITRAC PRIVATE -static)
  endif()
  add_custom_command(TARGET SITRAC POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/datos" "$<TARGET_FILE_DIR:SITRAC>/datos")
else()
  add_executable(sitrac src/main.cpp ${MODELO})
endif()
